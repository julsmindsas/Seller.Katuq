5. FLUJO DE SELECCIÓN Y PERSISTENCIA
====================================

    USER CLICK ON TOOL CARD
            |
            ↓
    toggleTool('getTotalSales')
            |
            ├─> if (toolName NOT in selectedTools)
            |   ├─> selectedTools.push('getTotalSales')
            |   └─> Update UI: add green check
            |
            └─> else
                ├─> selectedTools.splice(index, 1)
                └─> Update UI: remove green check
    
    
    onInputChange()
            |
            ↓
    agentConfigChange.emit(agentConfig)
            |
            |  [(agentConfig)]="agentConfig"  (two-way binding)
            |
            ↓
    WizardComponent.agentConfig UPDATED
            |
            ├─> agentConfig = {
            |       agentName: "Asistente Ventas",
            |       department: "sales",
            |       systemPrompt: "Eres un asistente...",
            |       selectedTools: [
            |         "getTotalSales",
            |         "getCustomers"
            |       ],
            |       model: "gemini-2.5-flash"
            |   }
            |
            └─> Data ready to persist


6. CREACIÓN DEL AGENTE (Paso 4 → Backend)
===========================================

    USER CLICK "CREAR AGENTE"
            |
            ↓
    WizardComponent.createAgent()
            |
            ├─> Validate agentConfig ✓
            |
            ├─> Create request:
            |   {
            |     agentName: "Asistente Ventas",
            |     department: "sales",
            |     systemPrompt: "Eres un asistente...",
            |     selectedTools: ["getTotalSales", "getCustomers"],
            |     description: "...",
            |     model: "gemini-2.5-flash"
            |   }
            |
            └─> agentService.createAgent(request)
                    |
                    ↓
                POST /v1/agent-builder/create
                    |
                    ├─ Body: CreateAgentRequest (JSON)
                    |
                    └─ Response: { success, agent, message }
                            |
                            ├─> if success
                            |   ├─ Notification: "Agente creado"
                            |   └─ Navigate: /agent-builder/library
                            |
                            └─> if error
                                └─ Notification: "Error"

8. AGRUPACIÓN POR DEPARTAMENTO
==============================

    SELECCIÓN: department = "sales"
            |
            ↓
    loadAvailableTools() {
        const deptTools = toolCatalog["sales"] ||= [];  // Get sales tools
        const genTools = toolCatalog.general ||= [];    // Get general tools
        
        availableTools = [
            ... deptTools,   // 10 sales tools
            ... genTools     // 3 general tools (for everyone)
        ];  // Total: 13 tools
    }
    
    VISTA AL USUARIO:
    - Tools de Ventas (10): getTotalSales, getCustomers, ...
    - Tools Generales (3): send_email, generate_report, ...
    - Total visible: 13 tools
    
    
    SI SELECCIONA: department = "inventory"
            |
            ↓
    loadAvailableTools() {
        const deptTools = toolCatalog["inventory"] ||= [];  // Get inventory tools
        const genTools = toolCatalog.general ||= [];        // Get general tools
        
        availableTools = [
            ... deptTools,   // 5 inventory tools
            ... genTools     // 3 general tools
        ];  // Total: 8 tools
    }
    
    VISTA AL USUARIO:
    - Tools de Inventario (5): getProductStock, checkLowStock, ...
    - Tools Generales (3): send_email, generate_report, ...
    - Total visible: 8 tools
